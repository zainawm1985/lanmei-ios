name: build-ipa

on:
  workflow_dispatch:
  push:
    branches: [main]

jobs:
  ipa:
    runs-on: macos-14
    timeout-minutes: 30
    steps:
      - uses: actions/checkout@v4

      - name: Install XcodeGen
        run: brew install xcodegen

      - name: Generate Xcode project
        run: xcodegen generate

      - name: Build (unsigned)
        run: |
          set -o pipefail
          xcodebuild \
            -project LanmeiVideo.xcodeproj \
            -scheme LanmeiVideo \
            -configuration Release \
            -sdk iphoneos \
            -derivedDataPath build \
            CODE_SIGN_IDENTITY="" \
            CODE_SIGNING_REQUIRED=NO \
            CODE_SIGNING_ALLOWED=NO \
            ENABLE_BITCODE=NO \
            build | tail -n 40

      - name: Package unsigned IPA
        run: |
          APP="build/Build/Products/Release-iphoneos/LanmeiVideo.app"
          test -d "$APP" || { echo "app not found"; exit 1; }
          rm -rf Payload LanmeiVideo-unsigned.ipa
          mkdir -p Payload
          cp -R "$APP" Payload/
          zip -qry LanmeiVideo-unsigned.ipa Payload
          ls -lh LanmeiVideo-unsigned.ipa

      - uses: actions/upload-artifact@v4
        with:
          name: LanmeiVideo-unsigned-ipa
          path: LanmeiVideo-unsigned.ipa
          retention-days: 30
